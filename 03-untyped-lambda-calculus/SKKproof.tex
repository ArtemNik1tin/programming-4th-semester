\noindent\textbf{Базовые правила редукции:}
\begin{align*}
    Sxyz &= xz(yz) \\
    Kxy &= x
\end{align*}

\noindent\textbf{Утверждение.} $SKK = I$

\begin{proof}[Доказательство]
Применим комбинатор $SKK$ к произвольному терму $x$:
\begin{align*}
    (SKK)x &= Kx(Kx) \quad \text{--- по правилу для } S \\
           &= x \quad\qquad \text{--- по правилу для } K
\end{align*}
Так как $(SKK)x = x$ для любого $x$, то по принципу экстенсиональности:
\[ SKK = I \]
\end{proof}
